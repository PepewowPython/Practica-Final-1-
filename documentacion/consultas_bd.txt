RF-01 / RI-01: GESTIÓN DE ROLES Y PERMISOS DE ACCESO (RBAC)
Tabla: roles
------------------------------------------------------------------------------
-- DDL: Creación de la tabla de roles
CREATE TABLE roles (
  id_rol INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del rol',
  nombre_rol VARCHAR(50) NOT NULL UNIQUE COMMENT 'Nombre del rol (ej: Administrador, Moderador, Usuario Ciudadano)',
  descripcion TEXT NULL COMMENT 'Descripción detallada de los permisos',
  nivel_acceso INT NOT NULL DEFAULT 1 COMMENT 'Nivel jerárquico (1: Usuario, 2: Moderador, 3: Admin)'
) ENGINE=InnoDB COMMENT='Tabla de catálogo de roles del sistema';

-- CONSULTA: Obtener todos los roles disponibles

RF-02 / RI-02: GESTIÓN DE USUARIOS Y AUTENTICACIÓN
Tabla: usuarios
------------------------------------------------------------------------------
-- DDL: Creación de la tabla de usuarios
CREATE TABLE usuarios (
  id_usuario INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único del usuario',
  nombre VARCHAR(100) NOT NULL COMMENT 'Nombre completo del usuario',
  correo VARCHAR(150) NOT NULL UNIQUE COMMENT 'Correo electrónico (Login ID)',
  contrasena VARCHAR(255) NOT NULL COMMENT 'Hash de contraseña (bcrypt)',
  telefono VARCHAR(20) NULL COMMENT 'Teléfono de contacto',
  estado ENUM('activo', 'inactivo', 'bloqueado') NOT NULL DEFAULT 'activo' COMMENT 'Estado de la cuenta',
  fecha_registro DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha y hora de creación',
  id_rol INT NOT NULL COMMENT 'Clave foránea hacia tabla Roles',
  CONSTRAINT fk_usuarios_roles FOREIGN KEY (id_rol) 
    REFERENCES roles(id_rol) ON DELETE RESTRICT ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Tabla principal de usuarios registrados';

Tabla: ubicaciones
------------------------------------------------------------------------------
-- DDL: Creación de la tabla de posiciones GPS
CREATE TABLE ubicaciones (
  id_ubicacion INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador de registro de ubicación',
  id_usuario INT NOT NULL COMMENT 'Usuario al que pertenece el registro',
  latitud DECIMAL(10,8) NOT NULL COMMENT 'Coordenada Latitud GPS',
  longitud DECIMAL(11,8) NOT NULL COMMENT 'Coordenada Longitud GPS',
  fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Timestamp de la captura',
  CONSTRAINT fk_ubicaciones_usuarios FOREIGN KEY (id_usuario) 
    REFERENCES usuarios(id_usuario) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Historial de geolocalización en tiempo real';

-- CONSULTA: Obtener la última posición conocida de un usuario
SELECT id_ubicacion, id_usuario, latitud, longitud, fecha_hora
FROM ubicaciones
WHERE id_usuario = 3
ORDER BY fecha_hora DESC
LIMIT 1;

-- CONSULTA: Consultar trayecto histórico de ubicaciones en un rango de fecha
SELECT latitud, longitud, fecha_hora
FROM ubicaciones
WHERE id_usuario = 3 AND fecha_hora >= DATE_SUB(NOW(), INTERVAL 1 DAY)
ORDER BY fecha_hora ASC;

RF-04 / RI-04: CÁLCULO DE RUTAS SEGUDRAS Y EVALUACIÓN DE RIESGO
Tabla: rutas
------------------------------------------------------------------------------
-- DDL: Creación de la tabla de rutas
CREATE TABLE rutas (
  id_ruta INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador de la consulta de ruta',
  id_usuario INT NOT NULL COMMENT 'Usuario que consultó la ruta',
  origen VARCHAR(255) NOT NULL COMMENT 'Dirección o punto de origen',
  destino VARCHAR(255) NOT NULL COMMENT 'Dirección o punto de destino',
  distancia DECIMAL(8,2) NOT NULL COMMENT 'Distancia total en kilómetros',
  duracion_estimada_min INT NOT NULL DEFAULT 0 COMMENT 'Tiempo estimado en minutos',

  fecha_consulta DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha del cálculo',
  CONSTRAINT fk_rutas_usuarios FOREIGN KEY (id_usuario) 
    REFERENCES usuarios(id_usuario) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Registro de rutas solicitadas y analizadas';

-- CONSULTA: Obtener historial de rutas solicitadas por un usuario

RF-06 / RI-06: CATALOGACIÓN Y VISUALIZACIÓN DE ZONAS DE RIESGO
Tabla: zonas_riesgo
------------------------------------------------------------------------------
-- DDL: Creación de la tabla de zonas de riesgo
CREATE TABLE zonas_riesgo (
  id_zona INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador único de la zona peligrosa',
  nombre_zona VARCHAR(150) NOT NULL COMMENT 'Nombre o barrio de la zona',
  nivel_riesgo ENUM('Bajo', 'Medio', 'Alto') NOT NULL DEFAULT 'Alto' COMMENT 'Nivel de peligro',
  latitud DECIMAL(10,8) NOT NULL COMMENT 'Latitud centro de la zona',
  longitud DECIMAL(11,8) NOT NULL COMMENT 'Longitud centro de la zona',
  radio_metros INT NOT NULL DEFAULT 500 COMMENT 'Radio de peligro en metros',
  coordenadas TEXT NULL COMMENT 'Geometría JSON o polígono',
  fecha_creacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de registro'
) ENGINE=InnoDB COMMENT='Catálogo de zonas geográficas clasificadas por riesgo';

-- DML: Insertar zonas de riesgo iniciales (Seed Data)
INSERT INTO zonas_riesgo (id_zona, nombre_zona, nivel_riesgo, latitud, longitud, radio_metros) VALUES
(1, 'Sector Parque de las Luces / San Antonio', 'Alto', 6.24530000, -75.56840000, 600),
(2, 'Sector Prado Centro Norte', 'Medio', 6.25700000, -75.56500000, 450),
(3, 'Alrededores Universidad de Antioquia (Noche)', 'Alto', 6.26290000, -75.56840000, 500);

-- CONSULTA: Obtener todas las zonas de riesgo para dibujar capas en el mapa
SELECT id_zona, nombre_zona, nivel_riesgo, latitud, longitud, radio_metros, fecha_creacion
FROM zonas_riesgo
ORDER BY nivel_riesgo DESC;

-- CONSULTA: Detectar si una coordenada GPS dada está cerca de una zona de riesgo
SELECT id_zona, nombre_zona, nivel_riesgo, latitud, longitud, radio_metros
FROM zonas_riesgo
WHERE (6371000 * ACOS(COS(RADIANS(6.2453)) * COS(RADIANS(latitud)) * COS(RADIANS(longitud) - RADIANS(-75.5684)) + SIN(RADIANS(6.2453)) * SIN(RADIANS(latitud)))) <= radio_metros;

RF-07 / RI-07: ACTUALIZACIÓN Y TRAZABILIDAD DEL NIVEL DE RIESGO

------------------------------------------------------------------------------
-- DDL: Creación de la tabla historial_riesgo
CREATE TABLE historial_riesgo (
  id_historial INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador del registro histórico',
  id_zona INT NOT NULL COMMENT 'Zona asociada',
  fecha_actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha del cambio',
  nivel_riesgo_anterior ENUM('Bajo', 'Medio', 'Alto') NULL COMMENT 'Nivel previo',
  nivel_riesgo_nuevo ENUM('Bajo', 'Medio', 'Alto') NOT NULL COMMENT 'Nuevo nivel',
  motivo VARCHAR(255) NULL COMMENT 'Razón del cambio',
  CONSTRAINT fk_historial_zonas FOREIGN KEY (id_zona) 
    REFERENCES zonas_riesgo(id_zona) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Histórico de modificaciones en los niveles de riesgo de zonas';

WHERE id_zona = 2;

VALUES (2, 'Medio', 'Alto', 'Incremento del 40% en reportes de hurto durante el último mes');

-- CONSULTA: Consultar trazabilidad de modificaciones de nivel de riesgo

  estado ENUM('pendiente', 'aprobado', 'rechazado') NOT NULL DEFAULT 'pendiente' COMMENT 'Estado del reporte',
  CONSTRAINT fk_incidentes_usuarios FOREIGN KEY (id_usuario) 
    REFERENCES usuarios(id_usuario) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Reportes ciudadanos de incidentes de inseguridad';

-- CONSULTA: Listar incidentes pendientes para la cola de moderación

SELECT id_incidente, tipo_incidente, descripcion, latitud, longitud, ubicacion, fecha_reporte
FROM incidentes
WHERE estado = 'aprobado'
ORDER BY fecha_reporte DESC;

-- DDL: Creación de la tabla ubicaciones_compartidas

  id_usuario INT NOT NULL COMMENT 'Usuario que comparte trayecto',
  id_contacto INT NOT NULL COMMENT 'Contacto receptor autorizado',
  fecha_inicio DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Inicio de seguimiento',
  fecha_fin DATETIME NULL COMMENT 'Fin de seguimiento',

  CONSTRAINT fk_compartido_usuario FOREIGN KEY (id_usuario) 
    REFERENCES usuarios(id_usuario) ON DELETE CASCADE ON UPDATE CASCADE,
  CONSTRAINT fk_compartido_contacto FOREIGN KEY (id_contacto) 
    REFERENCES contactos_confianza(id_contacto) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Sesiones activas e históricas de ubicación compartida';

-- DML: Finalizar sesión de monitoreo
UPDATE ubicaciones_compartidas 
SET fecha_fin = CURRENT_TIMESTAMP, estado = 'finalizada'
WHERE id_compartido = 1;

-- CONSULTA: Consultar sesiones de seguimiento activas con detalles del contacto
SELECT uc.id_compartido, u.nombre AS emisor, c.nombre_contacto AS receptor, c.telefono, uc.fecha_inicio, uc.estado
FROM ubicaciones_compartidas uc

-- DDL: Creación de la tabla auditoria
CREATE TABLE auditoria (
  id_auditoria INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador de evento de auditoría',
  id_usuario INT NULL COMMENT 'Usuario responsable (NULL si es proceso del sistema)',
  accion VARCHAR(100) NOT NULL COMMENT 'Operación ejecutada',
  detalles TEXT NULL COMMENT 'Detalles técnicos o JSON',
  fecha_hora DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Momento exacto',
  CONSTRAINT fk_auditoria_usuario FOREIGN KEY (id_usuario) 
    REFERENCES usuarios(id_usuario) ON DELETE SET NULL ON UPDATE CASCADE
) ENGINE=InnoDB COMMENT='Bitácora de seguridad y auditoría general del sistema';

-- TRIGGERS DE AUTOMATIZACIÓN DE AUDITORÍA
DELIMITER //

CREATE TRIGGER trg_auditoria_nuevo_usuario
AFTER INSERT ON usuarios
FOR EACH ROW
BEGIN
  INSERT INTO auditoria (id_usuario, accion, detalles)
  VALUES (NEW.id_usuario, 'REGISTRO_USUARIO', CONCAT('Usuario registrado: ', NEW.correo));
END;
//

CREATE TRIGGER trg_auditoria_nuevo_incidente
AFTER INSERT ON incidentes
FOR EACH ROW
BEGIN
  INSERT INTO auditoria (id_usuario, accion, detalles)
  VALUES (NEW.id_usuario, 'INCIDENTE_REPORTADO', CONCAT('Tipo: ', NEW.tipo_incidente, ' en ', NEW.ubicacion));
END;
//

DELIMITER ;

-- CONSULTA: Consultar la bitácora de auditoría del sistema
SELECT a.id_auditoria, COALESCE(u.correo, 'SISTEMA') AS ejecutado_por, a.accion, a.detalles, a.fecha_hora
FROM auditoria a
LEFT JOIN usuarios u ON a.id_usuario = u.id_usuario
ORDER BY a.fecha_hora DESC
LIMIT 100;

-- DDL: Creación de la tabla estadisticas
CREATE TABLE estadisticas (
  id_estadistica INT AUTO_INCREMENT PRIMARY KEY COMMENT 'Identificador del informe',
  tipo_reporte VARCHAR(100) NOT NULL COMMENT 'Nombre del reporte',
  fecha_generacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT 'Fecha de cálculo',
  resultado JSON NOT NULL COMMENT 'Métricas analíticas en formato JSON'
) ENGINE=InnoDB COMMENT='Informes consolidados y métricas de seguridad';

-- CONSULTA ANALÍTICA 1: Conteo de incidentes por categoría/tipo
SELECT tipo_incidente, COUNT(*) AS total_incidentes, MAX(fecha_reporte) AS ultimo_reporte
FROM incidentes
WHERE estado = 'aprobado'
GROUP BY tipo_incidente
ORDER BY total_incidentes DESC;

-- CONSULTA ANALÍTICA 2: Distribución de incidentes por franja horaria
SELECT 
  CASE 
    WHEN HOUR(fecha_reporte) BETWEEN 6 AND 11 THEN 'Mañana (06:00 - 11:59)'
    WHEN HOUR(fecha_reporte) BETWEEN 12 AND 17 THEN 'Tarde (12:00 - 17:59)'
    WHEN HOUR(fecha_reporte) BETWEEN 18 AND 23 THEN 'Noche (18:00 - 23:59)'
    ELSE 'Madrugada (00:00 - 05:59)'
  END AS franja_horaria,
  COUNT(*) AS total_incidentes
FROM incidentes
GROUP BY franja_horaria
ORDER BY total_incidentes DESC;

-- CONSULTA ANALÍTICA 3: Resumen general de incidentes por estado
SELECT 
  COUNT(*) AS total_registrados,
  SUM(CASE WHEN estado = 'aprobado' THEN 1 ELSE 0 END) AS aprobados,
  SUM(CASE WHEN estado = 'pendiente' THEN 1 ELSE 0 END) AS pendientes,
  SUM(CASE WHEN estado = 'rechazado' THEN 1 ELSE 0 END) AS rechazados
FROM incidentes;

-- CONSULTA DE INFORMES HISTÓRICOS GENERADOS
SELECT id_estadistica, tipo_reporte, fecha_generacion, resultado
FROM estadisticas
ORDER BY fecha_generacion DESC;
